cmake_minimum_required(VERSION 3.25)
project(nexus_ak_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
if(NOT DEFINED AK_ROOT)
    message(FATAL_ERROR "Set AK_ROOT to the patched SerenityOS source directory")
endif()
configure_file(${AK_ROOT}/AK/Debug.h.in ${CMAKE_CURRENT_BINARY_DIR}/AK/Debug.h @ONLY)
file(STRINGS ${AK_ROOT}/AK/CMakeLists.txt ak_source_lines REGEX "^    [A-Za-z0-9]+\\.cpp$")
foreach(line IN LISTS ak_source_lines)
    string(STRIP "${line}" filename)
    list(APPEND ak_sources "${AK_ROOT}/AK/${filename}")
endforeach()
add_library(ak_probe STATIC ${ak_sources})
target_include_directories(ak_probe PUBLIC ${CMAKE_CURRENT_BINARY_DIR} ${AK_ROOT})
target_compile_options(ak_probe PRIVATE -fno-exceptions -Wno-literal-suffix)
foreach(test_name IN ITEMS vector-patched-probe vector-alias-probe vector-rvalue-probe vector-insert-probe)
    add_executable(${test_name} ../${test_name}.cpp)
    target_compile_options(${test_name} PRIVATE -fno-exceptions -Wno-literal-suffix)
    target_compile_definitions(${test_name} PRIVATE NEXUS_AK_REAL_ASSERT)
    target_link_options(${test_name} PRIVATE -Wl,--wrap=malloc)
    target_link_libraries(${test_name} PRIVATE ak_probe pthread)
endforeach()
